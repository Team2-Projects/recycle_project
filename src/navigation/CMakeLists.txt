cmake_minimum_required(VERSION 3.8)
project(navigation)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(nav2_collision_monitor REQUIRED)
find_package(navigation_interface REQUIRED)
find_package(rclcpp REQUIRED)

ament_python_install_package(${PROJECT_NAME})
foreach(node auto_nav coverage_node recycle recycle_tracking_node)
  configure_file(scripts/node.in ${CMAKE_CURRENT_BINARY_DIR}/${node} @ONLY)
  install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
add_executable(tracking_collision_monitor src/tracking_collision_monitor.cpp)
target_compile_features(tracking_collision_monitor PUBLIC cxx_std_17)
ament_target_dependencies(tracking_collision_monitor nav2_collision_monitor navigation_interface rclcpp)
install(TARGETS tracking_collision_monitor DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(navigation_regression test
    APPEND_ENV PYTEST_DISABLE_PLUGIN_AUTOLOAD=1 PYTHONDONTWRITEBYTECODE=1)
endif()
ament_package()
